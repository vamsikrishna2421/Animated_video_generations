Which one for your PDF? A few pages, fits with room to spare: send it whole, with page numbers (Part 1). Long, or lots of questions: search by meaning and send the top 3 (Part 2). The code, runnable (Python; pip install pypdf transformers torch):
from pypdf import PdfReader
import torch
from transformers import AutoTokenizer, AutoModel
pages = [p.extract_text() for p in PdfReader("manual.pdf").pages]
name = "sentence-transformers/all-MiniLM-L6-v2"
tok, model = AutoTokenizer.from_pretrained(name), AutoModel.from_pretrained(name)
def embed(texts):
    b = tok(texts, padding=True, truncation=True, max_length=256, return_tensors="pt")
    with torch.no_grad(): h = model(**b).last_hidden_state
    m = b["attention_mask"].unsqueeze(-1).float()
    return torch.nn.functional.normalize((h * m).sum(1) / m.sum(1), dim=1)
page_vecs = embed(pages)   # every page, once
q_vec = embed([question])[0]
scores = (page_vecs @ q_vec).tolist()
top3 = sorted(range(len(pages)), key=lambda i: -scores[i])[:3]   # best first
doc = "\n\n".join(f"--- page {i+1} ---\n{pages[i]}" for i in top3)
RULES = 'Answer the question only from the document below. If the answer is not in it, say "Not in the document." Keep it to one or two short lines, and end with the page number, like (page 3).'
messages = [{"role": "system", "content": RULES}, {"role": "user", "content": f"Document:\n{doc}\n\nQuestion: {question}"}]
